#include "sdkconfig.h"

#include <stdio.h>
#include "driver/i2c_master.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"

#define SDA_PIN 21
#define SCL_PIN 22

void app_main(void)
{
    printf("\n");
    printf("=============================\n");
    printf("      I2C SENSOR TEST\n");
    printf("=============================\n\n");

    // I2C bus configureren
    i2c_master_bus_config_t bus_config = {
        .i2c_port = I2C_NUM_0,
        .sda_io_num = SDA_PIN,
        .scl_io_num = SCL_PIN,
        .clk_source = I2C_CLK_SRC_DEFAULT,
        .glitch_ignore_cnt = 7,
        .flags.enable_internal_pullup = true,
    };

    i2c_master_bus_handle_t bus_handle;

    ESP_ERROR_CHECK(i2c_new_master_bus(&bus_config, &bus_handle));

    printf("I2C scanner gestart...\n\n");

    for (uint8_t address = 1; address < 127; address++)
    {
        esp_err_t result = i2c_master_probe(
            bus_handle,
            address,
            100
        );

        if (result == ESP_OK)
        {
            printf("I2C apparaat gevonden op: 0x%02X\n", address);
        }
    }

    printf("\nScanner klaar.\n");

    while (1)
    {
        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}
